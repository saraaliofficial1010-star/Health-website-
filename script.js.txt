// 1. Random Tip Functionality
const tipButton = document.getElementById('randomTipButton');
const tipDisplay = document.getElementById('tipDisplay');

tipButton.addEventListener('click', () => {
    // Get a random index from the healthTips array
    const randomIndex = Math.floor(Math.random() * healthTips.length);
    
    // Display the random tip
    tipDisplay.innerHTML = healthTips[randomIndex];
});
